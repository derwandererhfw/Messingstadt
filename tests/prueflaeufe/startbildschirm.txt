# Startbildschirm: "Neues Spiel" und "Beenden" wirken, der Rest ist sichtbar, ausgegraut und mit Tasten nicht erreichbar.
# Aufruf mit --menu --fresh.
wait 20
expect screen Startbildschirm
expect focus Neues Spiel
expect hud False
expect help False
# Titelbild, Schriftzug und bewegte Szene sind eingebunden
expect title bild,schriftzug,szene
remember tick
wait 30
same tick
# Pfeiltasten springen über die ausgegrauten Einträge zwischen Neues Spiel und Beenden
key Down
expect focus Beenden
key Up
expect focus Neues Spiel
# Klicks auf ausgegraute Einträge bewirken nichts
clickany Spiel laden  (bald)
expect screen Startbildschirm
clickany Einstellungen  (bald)
expect screen Startbildschirm
clickany Mitwirkende  (bald)
expect screen Startbildschirm
# Esc verlässt den Startbildschirm nicht
key Escape
expect screen Startbildschirm
print
# Enter startet das Spiel
key Enter
wait 10
expect screen
expect hud True
remember tick
wait 30
changed tick
# Über die Pausenmappe zurück, dann mit der Maus wieder hinein
key Escape
click Zum Hauptmenü
click Zum Hauptmenü
wait 10
expect screen Startbildschirm
expect focus Neues Spiel
click Neues Spiel
wait 10
expect screen
expect hud True
